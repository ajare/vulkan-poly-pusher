cmake_minimum_required(VERSION 3.22)
project(FrozenMppCpuComparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MPP_REFERENCE_SOURCE "" CACHE PATH "Read-only pinned MPP source")
set(MPP_REFERENCE_LIBDIR "" CACHE PATH "Built frozen MPP shared libraries")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MPP_REFERENCE_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE reference_sha OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT reference_sha STREQUAL "21b1bcff040a4cfd32e3a5e814b9a4092d2b7cef")
    message(FATAL_ERROR "Comparison requires the immutable MPP migration reference, not ${reference_sha}")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MPP_REFERENCE_SOURCE}" diff --exit-code HEAD
    RESULT_VARIABLE reference_modified OUTPUT_QUIET)
if(NOT reference_modified EQUAL 0)
    message(FATAL_ERROR "Comparison reference has tracked source or dependency changes")
endif()
set(VPP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
foreach(module mpp mpp-data mpp-mesh mpp-program mpp-resource-parsers mpp-mesh-specification-parser mpp-helper mpp-app-support)
    include_directories("${MPP_REFERENCE_SOURCE}/${module}/include")
endforeach()
include_directories("${MPP_REFERENCE_SOURCE}/vendor/include" "${MPP_REFERENCE_SOURCE}/ext/utils/include")
foreach(target MppData MppMesh MppProgram MassivePolyPusher MppResourceParsers MppMeshSpecificationParser MppHelper Utils)
    add_library(${target} SHARED IMPORTED)
    set_target_properties(${target} PROPERTIES IMPORTED_LOCATION "${MPP_REFERENCE_LIBDIR}/${CMAKE_SHARED_LIBRARY_PREFIX}${target}${CMAKE_SHARED_LIBRARY_SUFFIX}")
endforeach()
add_executable(MppAssetValidate "${VPP_ROOT}/tools/AssetValidate.cpp")
target_link_libraries(MppAssetValidate PRIVATE MppResourceParsers MassivePolyPusher MppData MppProgram MppMesh)
add_library(MppPublicInterfaces OBJECT "${VPP_ROOT}/tests/PublicInterfaces.cpp")
# Baseline app support is static; compile its unchanged context-free sources here.
add_executable(MppCpuTests "${VPP_ROOT}/tests/CpuTests.cpp")
foreach(source BackgroundWork CommandStack DocumentFile DocumentFoundationTests DocumentId PackageManifest ZipArchive)
    target_sources(MppCpuTests PRIVATE "${MPP_REFERENCE_SOURCE}/mpp-app-support/src/${source}.cpp")
endforeach()
find_package(Threads REQUIRED)
target_link_libraries(MppCpuTests PRIVATE MppResourceParsers MassivePolyPusher MppData MppProgram MppMesh Utils Threads::Threads)
set(CMAKE_BUILD_RPATH "${MPP_REFERENCE_LIBDIR}")
option(MPP_REFERENCE_CLEAR "Build the shared clear/present comparison" ON)
if(MPP_REFERENCE_CLEAR)
    find_package(OpenGL REQUIRED)
    foreach(target SDL3 GLEW)
        add_library(${target} SHARED IMPORTED)
        set_target_properties(${target} PROPERTIES IMPORTED_LOCATION "${MPP_REFERENCE_LIBDIR}/${CMAKE_SHARED_LIBRARY_PREFIX}${target}${CMAKE_SHARED_LIBRARY_SUFFIX}")
    endforeach()
    add_executable(MppClearPresent "${VPP_ROOT}/tools/ClearPresent.cpp")
    target_include_directories(MppClearPresent PRIVATE "${MPP_REFERENCE_SOURCE}/ext/sdl/include" "${MPP_REFERENCE_SOURCE}/ext/glew/include" "${MPP_REFERENCE_SOURCE}/build/_deps/glew-2.3.1/include")
    foreach(source Window WindowSDL InputManager InputManagerSDL Timer TimerSDL)
        target_sources(MppClearPresent PRIVATE "${MPP_REFERENCE_SOURCE}/mpp-app-support/src/${source}.cpp")
    endforeach()
    target_compile_definitions(MppClearPresent PRIVATE GLEW_NO_GLU)
    target_link_libraries(MppClearPresent PRIVATE MassivePolyPusher MppData MppMesh MppProgram SDL3 GLEW OpenGL::GL)
endif()
